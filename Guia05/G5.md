## 4. `src/components/CarCard.tsx`

```tsx
import React from "react";
import {
  View,
  Text,
  Image,
  StyleSheet
} from "react-native";

import { Car } from "../types/Car";

interface Props {
  car: Car;
}

export default function CarCard({ car }: Props) {
  return (
    <View style={styles.card}>
      <Image
        source={car.image}
        style={styles.image}
      />

      <Text style={styles.title}>
        {car.brand}
      </Text>

      <Text>
        Modelo: {car.model}
      </Text>

      <Text>
        Año: {car.year}
      </Text>

      <Text style={styles.price}>
        {car.price}
      </Text>
    </View>
  );
}

const styles = StyleSheet.create({
  card: {
    backgroundColor: "#FFF",
    margin: 10,
    padding: 15,
    borderRadius: 10,
    elevation: 4
  },

  image: {
    width: "100%",
    height: 150,
    resizeMode: "cover",
    borderRadius: 10
  },

  title: {
    fontSize: 22,
    fontWeight: "bold",
    marginTop: 10
  },

  price: {
    color: "green",
    fontWeight: "bold",
    marginTop: 5
  }
});
```

<a id="app"></a>
## 5. `App.tsx`

> Se utiliza `SafeAreaView` de **`react-native-safe-area-context`** (no la de `react-native`, que está deprecada). Si el paquete no está instalado en el proyecto, agréguelo con:
>
> ```bash
> npm i react-native-safe-area-context
> ```

```tsx
import React from "react";
import {  FlatList, View
} from "react-native";
import { SafeAreaProvider } from "react-native-safe-area-context";
import CarCard from "./components/CarCard";
import { CARS } from "./data/cars";

export default function App() {
 
  return (
      <SafeAreaProvider>
            <FlatList
        data={CARS}
        keyExtractor={(item) => item.id}
        renderItem={({ item }) => (
          <CarCard car={item} />
        )}
      />
    </SafeAreaProvider>

    
  );
}
```

<a id="eas"></a>
## 6. `eas.json` (configuración de build)

```json
{
  "build": {
    "preview": {
      "distribution": "internal",
      "android": {
        "buildType": "apk"
      }
    },
    "production": {}
  }
}
```

> La propiedad `"buildType": "apk"` es necesaria cuando se desea generar un archivo APK descargable. De lo contrario, Expo genera un Android App Bundle (AAB), usado para Google Play Store.

<a id="comandos"></a>
## 7. Comandos usados en la guía

```bash
# Verificar instalación de npm
npm -v

# Instalar Expo CLI
npm install -g expo-cli

# Crear proyecto Expo con TypeScript
npx create-expo-app@latest cars --template blank-typescript

# Ejecutar la app
npx expo start

# Inicializar Git
git init
git add .
git commit -m "Versión inicial"

# Instalar EAS CLI
npm install -g eas-cli

# Login en Expo
eas login

# Verificar sesión
eas whoami

# Configurar EAS Build
eas build:configure

# Guardar configuración EAS
git add .
git commit -m "Configuración EAS Build"

# Generar APK
eas build -p android --profile preview
```
